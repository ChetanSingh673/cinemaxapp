7) By default Jenkins will run as Jenkins user and cannot access Kubernetes Cluster Nodes.
Lets verify;
be as a root user;
kubectl get nodes ----> You will see the nodes

Switch to Jenkins user:
sudo su - jenkins

Check:
kubectl get nodes
It will probably fail because Jenkins doesn't have your kubeconfig.

Switch back to root user and execute below commands;
sudo mkdir -p /var/lib/jenkins/.kube
sudo cp ~/.kube/config /var/lib/jenkins/.kube/config
sudo chown -R jenkins:jenkins /var/lib/jenkins/.kube

sudo su - jenkins
kubectl get nodes ----> You will see the response

Add Jenkins to Docker Group (Be as a root user and execute below commands)
sudo usermod -aG docker jenkins
sudo systemctl restart jenkins

Verify:
sudo su - jenkins

docker ps
Must work without sudo.

8) Install Pipeline Stage view plugin

9) Configure dockerhub credentials as dockerhub-creds in Jenkins with kind as username with password

10) Run the Pipeline Below
pipeline {
agent any

environment {
    DOCKERHUB_CREDS = 'dockerhub-creds'
    DOCKERHUB_USER  = 'kastrov'

    FRONTEND_IMAGE = 'kastrov/cinebook-frontend'
    AUTH_IMAGE     = 'kastrov/cinebook-auth'
    MOVIE_IMAGE    = 'kastrov/cinebook-movie'
    BOOKING_IMAGE  = 'kastrov/cinebook-booking'
    PAYMENT_IMAGE  = 'kastrov/cinebook-payment'
}

options {
    timestamps()
    disableConcurrentBuilds()
}

stages {

    stage('Checkout Source') {
        steps {
            git branch: 'main',
                url: 'https://github.com/KastroVKiran/cine-max-app.git'
        }
    }

    stage('Verify Tools') {
        steps {
            sh '''
                docker --version
                kubectl version --client
                aws --version
            '''
        }
    }

    stage('Build Docker Images') {
        steps {
            sh '''
                docker build -t ${FRONTEND_IMAGE}:latest ./frontend
                docker build -t ${AUTH_IMAGE}:latest ./auth-service
                docker build -t ${MOVIE_IMAGE}:latest ./movie-service
                docker build -t ${BOOKING_IMAGE}:latest ./booking-service
                docker build -t ${PAYMENT_IMAGE}:latest ./payment-service
            '''
        }
    }

    stage('Push Images To DockerHub') {
        steps {

            withCredentials([
                usernamePassword(
                    credentialsId: "${DOCKERHUB_CREDS}",
                    usernameVariable: 'DOCKER_USER',
                    passwordVariable: 'DOCKER_PASS'
                )
            ]) {

                sh '''
                    echo "$DOCKER_PASS" | docker login -u "$DOCKER_USER" --password-stdin

                    docker push ${FRONTEND_IMAGE}:latest
                    docker push ${AUTH_IMAGE}:latest
                    docker push ${MOVIE_IMAGE}:latest
                    docker push ${BOOKING_IMAGE}:latest
                    docker push ${PAYMENT_IMAGE}:latest

                    docker logout
                '''
            }
        }
    }

    stage('Deploy Kubernetes Resources') {
        steps {
            sh '''
                kubectl apply -f k8s/namespaces/namespace.yaml

                kubectl apply -f k8s/mongodb/storageclass.yaml

                kubectl apply -f k8s/mongodb/mongodb-statefulsets.yaml

                kubectl apply -f k8s/services/secrets.yaml

                kubectl apply -f k8s/services/deployments.yaml

                kubectl apply -f k8s/ingress.yaml
            '''
        }
    }

    stage('Rollout Restart') {
        steps {
            sh '''
                kubectl rollout restart deployment/frontend -n cinebook
                kubectl rollout restart deployment/auth-service -n cinebook
                kubectl rollout restart deployment/movie-service -n cinebook
                kubectl rollout restart deployment/booking-service -n cinebook
                kubectl rollout restart deployment/payment-service -n cinebook
            '''
        }
    }

    stage('Verify Rollout') {
        steps {
            sh '''
                kubectl rollout status deployment/frontend -n cinebook --timeout=300s
                kubectl rollout status deployment/auth-service -n cinebook --timeout=300s
                kubectl rollout status deployment/movie-service -n cinebook --timeout=300s
                kubectl rollout status deployment/booking-service -n cinebook --timeout=300s
                kubectl rollout status deployment/payment-service -n cinebook --timeout=300s
            '''
        }
    }

    stage('Show Application URL') {
        steps {
            sh '''
                echo "==============================="
                echo "Ingress Information"
                echo "==============================="

                kubectl get ingress -n cinebook

                echo ""
                echo "ALB DNS:"
                kubectl get ingress cinebook-ingress -n cinebook -o jsonpath='{.status.loadBalancer.ingress[0].hostname}'
                echo ""
            '''
        }
    }
}

post {

    success {
        echo 'Deployment completed successfully.'
    }

    failure {
        echo 'Pipeline failed.'
    }

    always {
        sh 'docker image prune -f || true'
    }
}


}

14) Since domain is configured, we need to update the ingress.yml file in gtihub repo.
Paste the below content in the existing ingress.yml file and then run the pipeline
Note: No need to delete the old ingress, as we are referring to the same old ingress name.

apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: cinebook-ingress
  namespace: cinebook
  annotations:
    alb.ingress.kubernetes.io/scheme: internet-facing
    alb.ingress.kubernetes.io/target-type: ip
    alb.ingress.kubernetes.io/listen-ports: |
      [{"HTTP":80},{"HTTPS":443}]
    alb.ingress.kubernetes.io/ssl-redirect: '443'
    alb.ingress.kubernetes.io/certificate-arn: arn:aws:acm:ap-south-1:560185625463:certificate/REPLACE-WITH-YOUR-CERTIFICATE-ID
spec:
  ingressClassName: alb
  rules:
    - host: learndevops01.click
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: frontend

                  number: 5000
